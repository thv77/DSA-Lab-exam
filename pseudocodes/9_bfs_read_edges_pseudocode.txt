PSEUDOCODE 9: BFS FROM VERTICES AND EDGES (QUEUE)

BEGIN
    DECLARE adj[20][20], visited[20], queue[20], seq[20], edges[400][2]
    n = 0
    e = 0
    cnt = 0
    built = false
    performed = false
    REPEAT
        DISPLAY menu: 1 Read graph, 2 Construct matrix, 3 Perform BFS, 4 Display, 5 Exit
        READ choice
        CASE choice OF
            1: READ n
               IF n < 1 OR n > 20 THEN
                   PRINT "Invalid number of vertices"
                   n = 0
               ELSE
                   READ e
                   IF e < 0 OR e > 400 THEN
                       PRINT "Invalid number of edges"
                       e = 0
                   ELSE
                       FOR i = 0 TO e-1
                           READ edges[i][0], edges[i][1]
                       END FOR
                       built = false
                       performed = false
                   END IF
               END IF
            2: IF n == 0 THEN
                   PRINT "Read vertices and edges first"
               ELSE
                   SET every adj[i][j] = 0
                   FOR i = 0 TO e-1
                       u = edges[i][0]
                       v = edges[i][1]
                       IF u or v is out of range THEN
                           PRINT "Skipping invalid edge"
                       ELSE
                           adj[u][v] = 1
                           adj[v][u] = 1
                       END IF
                   END FOR
                   built = true
                   performed = false
                   PRINT adjacency matrix
               END IF
            3: IF NOT built THEN
                   PRINT "Construct the adjacency matrix first"
               ELSE
                   READ start
                   IF start < 0 OR start >= n THEN
                       PRINT "Invalid vertex"
                   ELSE
                       FOR i = 0 TO n-1
                           visited[i] = 0
                       END FOR
                       front = 0
                       rear = 0
                       cnt = 0
                       visited[start] = 1
                       queue[rear] = start
                       rear = rear + 1
                       WHILE front < rear
                           v = queue[front]
                           front = front + 1
                           seq[cnt] = v
                           cnt = cnt + 1
                           FOR i = 0 TO n-1
                               IF adj[v][i] == 1 AND visited[i] == 0 THEN
                                   visited[i] = 1
                                   queue[rear] = i
                                   rear = rear + 1
                               END IF
                           END FOR
                       END WHILE
                       performed = true
                   END IF
               END IF
            4: IF NOT performed THEN
                   PRINT "Perform BFS first"
               ELSE
                   PRINT seq[0..cnt-1]
               END IF
            5: EXIT
            ELSE: PRINT "Invalid choice"
        END CASE
    UNTIL choice == 5
END
