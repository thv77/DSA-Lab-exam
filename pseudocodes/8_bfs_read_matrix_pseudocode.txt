PSEUDOCODE 8: BFS USING ADJACENCY MATRIX (QUEUE)

BEGIN
    DECLARE adj[20][20], visited[20], queue[20], seq[20]
    n = 0
    start = -1
    haveMatrix = false
    REPEAT
        DISPLAY menu: 1 Vertices, 2 Matrix, 3 Start vertex, 4 Display BFS, 5 Exit
        READ choice
        CASE choice OF
            1: READ n
               haveMatrix = false
               IF n < 1 OR n > 20 THEN
                   PRINT "Invalid number of vertices"
                   n = 0
               END IF
            2: IF n == 0 THEN
                   PRINT "Read vertices first"
               ELSE
                   FOR i = 0 TO n-1
                       FOR j = 0 TO n-1
                           READ adj[i][j]
                       END FOR
                   END FOR
                   haveMatrix = true
               END IF
            3: IF n == 0 THEN
                   PRINT "Read vertices first"
               ELSE
                   READ start
               END IF
            4: IF n == 0 OR NOT haveMatrix OR start < 0 OR start >= n THEN
                   PRINT "Provide vertices, matrix and valid start first"
               ELSE
                   FOR i = 0 TO n-1
                       visited[i] = 0
                   END FOR
                   front = 0
                   rear = 0
                   cnt = 0
                   visited[start] = 1
                   queue[rear] = start
                   rear = rear + 1
                   WHILE front < rear
                       v = queue[front]
                       front = front + 1
                       seq[cnt] = v
                       cnt = cnt + 1
                       FOR i = 0 TO n-1
                           IF adj[v][i] == 1 AND visited[i] == 0 THEN
                               visited[i] = 1
                               queue[rear] = i
                               rear = rear + 1
                           END IF
                       END FOR
                   END WHILE
                   PRINT seq[0..cnt-1]
               END IF
            5: EXIT
            ELSE: PRINT "Invalid choice"
        END CASE
    UNTIL choice == 5
END
