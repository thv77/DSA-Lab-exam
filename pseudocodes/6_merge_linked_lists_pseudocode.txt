PSEUDOCODE 6: MERGE TWO SINGLY LINKED LISTS

BEGIN
    head[0] = NULL
    head[1] = NULL
    FOR w = 0 TO 1
        READ n
        FOR i = 1 TO n
            READ v
            CREATE newnode with data = v, next = NULL
            IF head[w] == NULL OR v < head[w].data THEN
                newnode.next = head[w]
                head[w] = newnode
            ELSE
                p = head[w]
                WHILE p.next != NULL AND p.next.data <= v
                    p = p.next
                END WHILE
                newnode.next = p.next
                p.next = newnode
            END IF
        END FOR
    END FOR
